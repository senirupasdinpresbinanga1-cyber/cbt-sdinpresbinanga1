/** * GOOGLE APPS SCRIPT * Admin: senirupasdinpresbinanga1@gmail.com * SD INPRES BINANGA 1 MAMUJU * * Fitur: * 1. Menerima Data Nilai Ujian CBT dari Siswa & Menyimpan ke Spreadsheet * 2. Menyediakan API Hapus Data Per Baris maupun Hapus/Reset Seluruh Nilai */ function doPost(e) { var lock = LockService.getScriptLock(); lock.tryLock(10000); try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); // Inisialisasi Header bila sheet masih kosong if (sheet.getLastRow() === 0) { sheet.appendRow(["WAKTU DITERIMA", "NAMA SISWA", "NISN", "KELAS", "SKOR PG (MAX 240)", "JAWABAN URAIAN"]); sheet.getRange(1, 1, 1, 6).setFontWeight("bold").setBackground("#1677ff").setFontColor("white"); } var data = JSON.parse(e.postData.contents); // BILA AKSI "SAVE" -> Simpan Nilai Baru if (data.action === "save" || !data.action) { var timestamp = new Date(); sheet.appendRow([ timestamp, data.nama, "'" + data.nisn, // Menjaga format String NISN data.kelas, data.skorPG, data.jawabanUraian ]); return ContentService.createTextOutput(JSON.stringify({ "result": "success", "message": "Data berhasil disimpan!" })) .setMimeType(ContentService.MimeType.JSON); } // BILA AKSI "DELETE_ROW" -> Hapus Data Berdasarkan NISN else if (data.action === "delete_row") { var targetNisn = String(data.nisn); var rows = sheet.getDataRange().getValues(); var deleted = false; // Iterasi mundur untuk menghapus baris dengan aman for (var i = rows.length - 1; i >= 1; i--) { var rowNisn = String(rows[i][2]).replace("'", ""); if (rowNisn === targetNisn) { sheet.deleteRow(i + 1); deleted = true; } } if (deleted) { return ContentService.createTextOutput(JSON.stringify({ "result": "success", "message": "Data NISN " + targetNisn + " berhasil dihapus!" })) .setMimeType(ContentService.MimeType.JSON); } else { return ContentService.createTextOutput(JSON.stringify({ "result": "not_found", "message": "NISN tidak ditemukan!" })) .setMimeType(ContentService.MimeType.JSON); } } // BILA AKSI "CLEAR_ALL" -> Hapus Seluruh Data Nilai Siswa else if (data.action === "clear_all") { var lastRow = sheet.getLastRow(); if (lastRow > 1) { sheet.deleteRows(2, lastRow - 1); } return ContentService.createTextOutput(JSON.stringify({ "result": "success", "message": "Seluruh data nilai berhasil dibersihkan!" })) .setMimeType(ContentService.MimeType.JSON); } } catch (err) { return ContentService.createTextOutput(JSON.stringify({ "result": "error", "error": err.toString() })) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } function doGet(e) { return ContentService.createTextOutput("Google Apps Script CBT SD Inpres Binanga 1 Aktif!"); }
